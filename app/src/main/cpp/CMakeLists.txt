cmake_minimum_required(VERSION 3.22.1)
project(fx991cnx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
file(GLOB CHIP core/Chipset/*.cpp)
file(GLOB PERIPHERALS core/Peripheral/*.cpp)
add_library(fx991cnx SHARED bridge.cpp ${CHIP} ${PERIPHERALS})
target_include_directories(fx991cnx PRIVATE core)
target_compile_options(fx991cnx PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_options(fx991cnx PRIVATE "-Wl,-z,max-page-size=16384")
